``` sse_headers
Content-Type: text/event-stream
```

``` transform_stream
stream.mode = "transform";
stream.onEvent = (event) => ({ event: event.event, data: event.data });
stream.onEnd = () => ({ event: "done", data: "end" });
```

``` mock_stream
stream.mode = "mock";
let step = 0;
stream.next = () => {
  step += 1;
  if (step <= 3) {
    return {
      output: { event: "mock", data: String(step) },
      delayMs: 250,
      done: step === 3,
    };
  }
  return { done: true };
};
```

``` event_error_stream

stream.onEvent = () => { throw new Error("event boom"); };
```

``` end_error_stream
stream.mode = "transform";

stream.onEnd = () => { throw new Error("end boom"); };
```

``` mock_error_stream
stream.mode = "mock";
stream.next = () => { throw new Error("next boom"); };
```
